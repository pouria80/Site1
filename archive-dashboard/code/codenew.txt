import { useCallback, useEffect, useState, type ReactNode } from 'react';
import { motion } from 'framer-motion';
import { STR, type Dict } from '../i18n';
import type { Currency, JourneyItem, Lang, Scenario, Theme, Ticket } from '../types';
import { SEED_JOURNEY, TICKETS } from '../data/mock';
import Sidebar from '../components/Sidebar';
import TopBar from '../components/TopBar';
import NowSignal from '../components/NowSignal';
import WalletCard from '../components/WalletCard';
import QuickActions from '../components/QuickActions';
import YourWorld from '../components/YourWorld';
import Discovery from '../components/Discovery';
import Journey from '../components/Journey';
import Support from '../components/Support';
import Toast from '../components/Toast';
import { AddFundsModal, TicketModal } from '../components/Modal';

const EASE = [0.22, 0.9, 0.26, 1] as [number, number, number, number];

const Sec = ({ children, delay = 0, id }: { children: ReactNode; delay?: number; id?: string }) => (
  <motion.section id={id} initial={{ opacity: 0, y: 28 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.7, delay, ease: EASE }}>
    {children}
  </motion.section>
);

export default function CustomerHub() {
  const [lang, setLang] = useState<Lang>('fa');
  const [theme, setTheme] = useState<Theme>('dark');
  const [collapsed, setCollapsed] = useState(false);
  const [mobOpen, setMobOpen] = useState(false);
  const [scenario, setScenario] = useState<Scenario>('active');
  const [active, setActive] = useState('hub');
  const [cur, setCur] = useState<Currency>('USD');
  const [balance, setBalance] = useState(124.5);
  const [saved, setSaved] = useState<string[]>(['tg', 'dota', 'spot']);
  const [q, setQ] = useState('');
  const [orderOpen, setOrderOpen] = useState(false);
  const [modal, setModal] = useState<null | 'funds' | 'ticket'>(null);
  const [amt, setAmt] = useState(20);
  const [journey, setJourney] = useState<JourneyItem[]>(SEED_JOURNEY);
  const [tickets, setTickets] = useState<Ticket[]>(TICKETS);
  const [toast, setToast] = useState<string | null>(null);

  const t: Dict = STR[lang];
  const fa = lang === 'fa';

  useEffect(() => {
    document.documentElement.dir = fa ? 'rtl' : 'ltr';
    document.documentElement.lang = lang;
  }, [lang, fa]);

  useEffect(() => {
    if (!toast) return;
    const id = setTimeout(() => setToast(null), 2600);
    return () => clearTimeout(id);
  }, [toast]);

  useEffect(() => {
    const h = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        setModal(null);
        setMobOpen(false);
      }
    };
    window.addEventListener('keydown', h);
    return () => window.removeEventListener('keydown', h);
  }, []);

  const say = useCallback((m: string) => setToast(m), []);
  const scroll = (id: string) => document.getElementById(id)?.scrollIntoView({ behavior: 'smooth', block: 'start' });

  const nav = (key: string, anchor?: string) => {
    setActive(key);
    setMobOpen(false);
    if (anchor) scroll(anchor);
    else say(t.toastSection);
  };

  const toggleSave = (id: string) => setSaved((s) => (s.includes(id) ? s.filter((x) => x !== id) : [...s, id]));

  const addFunds = (a: number) => {
    setBalance((b) => b + a);
    setJourney((j) => [{ type: 'charge', k: 'jCharge', amt: `+$${a}.00`, time: { en: t.justNow, fa: t.justNow } }, ...j]);
    setModal(null);
    say(t.toastAdded(a));
  };

  const SCENARIOS: [Scenario, string][] = [
    ['active', t.scActive],
    ['clear', t.scClear],
    ['failed', t.scFailed],
    ['verify', t.scVerify],
  ];

  return (
    <div className="hub" data-theme={theme} data-lang={lang} dir={fa ? 'rtl' : 'ltr'}>
      <div className="atmo" />
      <div className="grain" />

      <div className="shell">
        <Sidebar
          t={t}
          lang={lang}
          theme={theme}
          active={active}
          collapsed={collapsed}
          mobOpen={mobOpen}
          onNav={nav}
          onLang={setLang}
          onTheme={() => setTheme((th) => (th === 'dark' ? 'light' : 'dark'))}
          onCollapse={() => setCollapsed((c) => !c)}
        />
        <div className={`backdrop ${mobOpen ? 'on' : ''}`} onClick={() => setMobOpen(false)} />

        <main className="main" style={{ order: 1 }}>
          <TopBar
            t={t}
            theme={theme}
            q={q}
            onQ={setQ}
            onTheme={() => setTheme((th) => (th === 'dark' ? 'light' : 'dark'))}
            onMenu={() => setMobOpen(true)}
            onEnter={() => scroll('discSec')}
            onProfile={() => say(t.toastSection)}
          />

          <div className="content">
            <Sec delay={0}>
              <div className="proto">
                <span>{t.proto}</span>
                {SCENARIOS.map(([k, lb]) => (
                  <button key={k} className={`pchip ${scenario === k ? 'on' : ''}`} onClick={() => setScenario(k)}>
                    {lb}
                  </button>
                ))}
              </div>
            </Sec>

            <Sec delay={0.06} id="nowSec">
              <NowSignal
                t={t}
                lang={lang}
                scenario={scenario}
                orderOpen={orderOpen}
                onToggleOrder={() => setOrderOpen((o) => !o)}
                onExplore={() => scroll('discSec')}
                onFix={() => say(scenario === 'failed' ? t.toastCheckout : t.toastSection)}
              />
            </Sec>

            <Sec delay={0.12} id="walletSec">
              <div className="wrow">
                <WalletCard
                  t={t}
                  lang={lang}
                  cur={cur}
                  balance={balance}
                  onCur={setCur}
                  onAdd={() => setModal('funds')}
                  onWithdraw={() => say(t.toastWithdraw)}
                />
                <QuickActions
                  t={t}
                  onAdd={() => setModal('funds')}
                  onWithdraw={() => say(t.toastWithdraw)}
                  onOrders={() => scroll('nowSec')}
                />
              </div>
            </Sec>

            <Sec delay={0.18} id="worldSec">
              <YourWorld t={t} lang={lang} saved={saved} onToggleSave={toggleSave} onContinue={() => say(t.toastCheckout)} />
            </Sec>

            <Sec delay={0.24} id="discSec">
              <Discovery t={t} lang={lang} q={q} scenario={scenario} saved={saved} onToggleSave={toggleSave} onView={() => say(t.toastCheckout)} />
            </Sec>

            <Sec delay={0.3} id="journeySec">
              <div className="jrrow">
                <Journey t={t} lang={lang} journey={journey} />
                <Support
                  t={t}
                  lang={lang}
                  tickets={tickets}
                  onNewTicket={() => setModal('ticket')}
                  onHelp={() => say(t.toastSection)}
                />
              </div>
            </Sec>

            <div className="foot">PooriTel Hub · Prototype 0.9 · {t.user}</div>
          </div>
        </main>
      </div>

      <AnimateModals
        modal={modal}
        t={t}
        amt={amt}
        setAmt={setAmt}
        onClose={() => setModal(null)}
        onFunds={() => addFunds(amt)}
        onTicket={() => {
          setTickets((x) => [...x, { id: '#PT-' + (9013 + x.length), t: { en: 'New ticket', fa: 'تیکت جدید' } }]);
          setModal(null);
          say(t.toastTicket);
        }}
      />

      <Toast msg={toast} />
    </div>
  );
}

function AnimateModals({
  modal,
  t,
  amt,
  setAmt,
  onClose,
  onFunds,
  onTicket,
}: {
  modal: null | 'funds' | 'ticket';
  t: Dict;
  amt: number;
  setAmt: (a: number) => void;
  onClose: () => void;
  onFunds: () => void;
  onTicket: () => void;
}) {
  return (
    <>
      {modal === 'funds' && <AddFundsModal t={t} amt={amt} setAmt={setAmt} onConfirm={onFunds} onClose={onClose} />}
      {modal === 'ticket' && <TicketModal t={t} onSubmit={onTicket} onClose={onClose} />}
    </>
  );
}

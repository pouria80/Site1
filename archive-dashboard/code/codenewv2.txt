const EASE = [0.22, 0.9, 0.26, 1];
const Sec = ({ children, delay = 0, id }) => (
  <motion.section id={id} initial={{ opacity: 0, y: 28 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.7, delay, ease: EASE }}>
    {children}
  </motion.section>

function Particles() {
  const ref = useRef(null);
  useEffect(() => {
    const cv = ref.current; if (!cv) return;
    const ctx = cv.getContext('2d');
    const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
    let w, h, raf, ps = [];
    const dpr = Math.min(2, window.devicePixelRatio || 1);
    const resize = () => { w = cv.width = cv.offsetWidth * dpr; h = cv.height = cv.offsetHeight * dpr; };
    resize();
    ps = Array.from({ length: 34 }, () => ({
      x: Math.random() * w, y: Math.random() * h, r: (Math.random() * 1.3 + 0.4) * dpr,
      vx: (Math.random() - 0.5) * 0.12 * dpr, vy: -(Math.random() * 0.16 + 0.04) * dpr,
      a: Math.random() * 0.45 + 0.12, tw: Math.random() * Math.PI * 2,
    }));
    const draw = () => {
      ctx.clearRect(0, 0, w, h);
      for (const p of ps) {
        p.x += p.vx; p.y += p.vy; p.tw += 0.02;
        if (p.y < -6) p.y = h + 6; if (p.x < -6) p.x = w + 6; if (p.x > w + 6) p.x = -6;
        ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, 7);
        ctx.fillStyle = `rgba(232,178,80,${p.a * (0.6 + 0.4 * Math.sin(p.tw))})`; ctx.fill();
      }
      raf = requestAnimationFrame(draw);
    };
    if (!reduce) draw();
    const ro = new ResizeObserver(resize); ro.observe(cv);
    return () => { cancelAnimationFrame(raf); ro.disconnect(); };
  }, []);
  return <canvas ref={ref} className="pfield" />;
}

                  <h3><i />{t.wContinue}</h3>
                  {CONTINUE.map((c, i) => (
                    <button key={i} className="witem" style={{ alignItems: 'flex-start' }} onClick={() => say(t.toastCheckout)}>
                      <span className="wt">
                        <b>{c.t[lang]}</b><span>{c.s[lang]}</span>
                        <span className="pbar"><i style={{ width: c.pct + '%' }} /></span>
                      </span>
                      <span className="go" style={{ opacity: 1, transform: 'none' }}><I.chev size={14} className="flip" /></span>
                    </button>
                  ))}
                </div>
              </div>
            </Sec>

            {/* ============ Discover ============ */}
            <Sec delay={0.24} id="discSec">
              <div className="disc-head">
                <div>
                  <div className="hud">{t.discoverTitle}</div>
                  <div className="sub" style={{ marginTop: 8 }}>{q ? t.resultsFor(products.length) : (scenario === 'clear' ? t.becauseExplored : t.pickedForYou)}</div>
                </div>
              </div>
              {products.length === 0 ? (
                <div className="wcard" style={{ padding: 30, textAlign: 'center', color: 'var(--dim)' }}>{t.noResults}</div>
              ) : (
                <div className="disc">
                  {products.map((p, i) => (
                    <motion.button key={p.id} className="pcard" initial={{ opacity: 0, y: 22 }} whileInView={{ opacity: 1, y: 0 }} viewport={{ once: true }} transition={{ duration: 0.55, delay: i * 0.07, ease: EASE }} onClick={() => say(t.toastCheckout)}>
                      <div className="pmedia">
                        <span className="ctx">{CTX[p.ctx][lang]}</span>
                        <motion.span className={`fav ${saved.includes(p.id) ? 'on' : ''}`} whileTap={{ scale: 0.75 }}
                          onClick={e => { e.stopPropagation(); toggleSave(p.id); }}>
                          <I.heart size={15} />
                        </motion.span>
                        <img src={p.img} alt={p.name.en} loading="lazy" />
                      </div>
                      <div className="pinfo">
                        <div className="pname">{p.name[lang]}</div>
                        <div className="psub">{p.sub[lang]}</div>
                        <div className="prow">
                          <span className="price">{fmtUSD(p.price)}</span>
                          <span className="pview">{t.viewCta}<span className="flip"><I.chev size={11} /></span></span>
                        </div>
                      </div>
                    </motion.button>
                  ))}
                </div>
              )}
            </Sec>

            {/* ============ Journey + Support ============ */}
            <Sec delay={0.3} id="journeySec">
              <div className="jrrow">
                <div className="jcard">
                  <div className="hud" style={{ marginBottom: 18 }}>{t.journeyTitle}</div>
                  {journey.map((j, i) => (
                    <div key={i} className={`jitem ${i === 0 ? 'hot' : ''}`}>
                      {i < journey.length - 1 && <span className="jline" />}
                      <span className={`jnode ${j.type}`}>
                        {j.type === 'charge' ? <I.zap size={15} /> : j.type === 'order' ? <I.box size={15} /> : j.type === 'view' ? <I.eye size={15} /> : <I.shield size={15} />}
                      </span>
                      <span className="jbody"><b>{t[j.k]}</b><span>{j.time[lang]}{j.tagK ? ' · ' + t[j.tagK] : ''}</span></span>
                      {j.amt && <span className={`jamt ${j.amt.startsWith('+') ? 'plus' : 'minus'}`}>{j.amt}</span>}
                    </div>
                  ))}
                </div>
                <div className="sup" id="supportSec">
                  <div className="hud">{t.helpTitle}</div>
                  <p>{t.helpSub}</p>
                  <div className="sup-btns">
                    <button className="btn btn-p" onClick={() => setModal('ticket')}><I.ticket size={14} />{t.newTicket}</button>
                    <button className="btn btn-g" onClick={() => say(t.toastSection)}><I.book size={14} />{t.helpCenter}</button>
                  </div>
                  <div className="tchips">
                    {tickets.map((tk, i) => <span key={i} className="tchip"><i />{tk.id} · {tk.t[lang]}</span>)}
                  </div>
                  <div className="bsub" style={{ marginTop: 'auto' }}>{t.openTickets(tickets.length)}</div>
                </div>
              </div>
            </Sec>

            <div className="foot">PooriTel Hub · Prototype 0.9 · {fa ? 'پوریا' : 'Pouria'}</div>
          </div>
        </main>
      </div>

      {/* ============ Modals ============ */}
      <AnimatePresence>
        {modal === 'funds' && (
          <motion.div className="mwrap" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} onClick={() => setModal(null)}>
            <motion.div className="modal" initial={{ y: 26, scale: 0.97 }} animate={{ y: 0, scale: 1 }} exit={{ y: 20, opacity: 0 }} onClick={e => e.stopPropagation()}>
              <h3>{t.modalAddTitle}</h3><div className="ms">{t.modalAddSub}</div>
              <div className="amts">{[10, 20, 50, 100].map(a => (
                <button key={a} className={`amt ${amt === a ? 'on' : ''}`} onClick={() => setAmt(a)}>${a}</button>))}
              </div>
              <div className="mrow">
                <button className="btn btn-g" onClick={() => setModal(null)}>{t.cancel}</button>
                <button className="btn btn-p" onClick={() => addFunds(amt)}><I.zap size={14} />{t.confirm}</button>
              </div>
            </motion.div>
          </motion.div>
        )}
        {modal === 'ticket' && (
          <motion.div className="mwrap" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} onClick={() => setModal(null)}>
            <motion.div className="modal" initial={{ y: 26, scale: 0.97 }} animate={{ y: 0, scale: 1 }} exit={{ y: 20, opacity: 0 }} onClick={e => e.stopPropagation()}>
              <h3>{t.modalTicketTitle}</h3><div className="ms">{t.modalTicketSub}</div>
              <input className="mfield" placeholder={t.subjectPh} />
              <textarea className="mfield" placeholder={t.msgPh} />
              <div className="mrow">
                <button className="btn btn-g" onClick={() => setModal(null)}>{t.cancel}</button>
                <button className="btn btn-p" onClick={() => { setTickets(x => [...x, { id: '#PT-' + (9013 + x.length), t: { en: 'New ticket', fa: 'تیکت جدید' } }]); setModal(null); say(t.toastTicket); }}>
                  <I.check size={14} />{t.submit}
                </button>
              </div>
            </motion.div>
          </motion.div>
        )}
      </AnimatePresence>

      <AnimatePresence>
        {toast && <motion.div className="toast" initial={{ opacity: 0, y: 16 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: 10 }}>{toast}</motion.div>}
      </AnimatePresence>
    </div>
  );
}
